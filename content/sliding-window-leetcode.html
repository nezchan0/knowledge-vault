643. Maximum Average Subarray I

Fixed Window

Given nums and integer k, return the maximum average of any contiguous subarray of length k.

Python
class Solution:
    def findMaxAverage(self, nums, k):
        window_sum = sum(nums[:k])
        max_sum = window_sum

        for i in range(k, len(nums)):
            window_sum += nums[i] - nums[i - k]
            max_sum = max(max_sum, window_sum)

        return max_sum / k
Time: O(n)
Space: O(1)

1423. Maximum Points You Can Obtain from Cards

Fixed Window (Complement)

Given cardPoints, pick exactly k cards from either end to maximize the sum of picked cards.

Python
class Solution:
    def maxScore(self, cardPoints, k):
        n = len(cardPoints)
        total = sum(cardPoints)
        window_size = n - k

        if window_size == 0:
            return total

        window_sum = sum(cardPoints[:window_size])
        min_window = window_sum

        for i in range(window_size, n):
            window_sum += cardPoints[i] - cardPoints[i - window_size]
            min_window = min(min_window, window_sum)

        return total - min_window
Time: O(n)
Space: O(1)

1456. Maximum Number of Vowels in a Substring of Given Length

Fixed Window

Given string s and integer k, return the maximum number of vowels in any substring of length k.

Python
class Solution:
    def maxVowels(self, s, k):
        vowels = set('aeiou')

        count = sum(1 for c in s[:k] if c in vowels)
        max_count = count

        for i in range(k, len(s)):
            if s[i] in vowels:
                count += 1
            if s[i - k] in vowels:
                count -= 1

            max_count = max(max_count, count)

        return max_count
Time: O(n)
Space: O(1)

567. Permutation in String

Fixed Window + Frequency

Return true if s2 contains a permutation of s1 as a contiguous substring.

Python
class Solution:
    def checkInclusion(self, s1, s2):
        n1, n2 = len(s1), len(s2)

        if n1 > n2:
            return False

        need = [0] * 26
        window = [0] * 26

        for c in s1:
            need[ord(c) - ord('a')] += 1

        for i in range(n2):
            window[ord(s2[i]) - ord('a')] += 1

            if i >= n1:
                window[ord(s2[i - n1]) - ord('a')] -= 1

            if window == need:
                return True

        return False
Time: O(n)
Space: O(1) — fixed 26-size arrays

438. Find All Anagrams in a String

Fixed Window + Frequency

Given s and p, return all start indices of p's anagrams in s.

Python
class Solution:
    def findAnagrams(self, s, p):
        ns, np = len(s), len(p)

        if np > ns:
            return []

        need = [0] * 26
        window = [0] * 26

        for c in p:
            need[ord(c) - ord('a')] += 1

        res = []

        for i in range(ns):
            window[ord(s[i]) - ord('a')] += 1

            if i >= np:
                window[ord(s[i - np]) - ord('a')] -= 1

            if window == need:
                res.append(i - np + 1)

        return res
Time: O(n)
Space: O(1) — fixed 26-size arrays

209. Minimum Size Subarray Sum

Variable Window

Given target and nums, return the minimal length of a contiguous subarray with sum ≥ target, or 0.

Python
class Solution:
    def minSubArrayLen(self, target, nums):
        left = 0
        curr_sum = 0
        min_len = float('inf')

        for right in range(len(nums)):
            curr_sum += nums[right]

            while curr_sum >= target:
                min_len = min(min_len, right - left + 1)
                curr_sum -= nums[left]
                left += 1

        return min_len if min_len != float('inf') else 0
Time: O(n)

3. Longest Substring Without Repeating Characters

Variable Window

Given a string s, return the length of the longest substring without repeating characters.

Python
class Solution:
    def lengthOfLongestSubstring(self, s):
        seen = {}
        left = 0
        max_len = 0

        for right, c in enumerate(s):
            if c in seen and seen[c] >= left:
                left = seen[c] + 1

            seen[c] = right
            max_len = max(max_len, right - left + 1)

        return max_len
Time: O(n)

1004. Max Consecutive Ones III

Variable Window

Given a binary array nums and integer k, return the max length of consecutive 1s after flipping at most k 0s.

Python
class Solution:
    def longestOnes(self, nums, k):
        left = 0
        zeros = 0
        max_len = 0

        for right in range(len(nums)):
            if nums[right] == 0:
                zeros += 1

            while zeros > k:
                if nums[left] == 0:
                    zeros -= 1
                left += 1

            max_len = max(max_len, right - left + 1)

        return max_len

424. Longest Repeating Character Replacement

Variable Window

Given string s and integer k, return the length of the longest substring of a single repeated letter after replacing at most k characters.

Python
class Solution:
    def characterReplacement(self, s, k):
        count = [0] * 26
        left = 0
        max_count = 0
        max_len = 0

        for right in range(len(s)):
            count[ord(s[right]) - ord('A')] += 1
            max_count = max(max_count, count[ord(s[right]) - ord('A')])

            while (right - left + 1) - max_count > k:
                count[ord(s[left]) - ord('A')] -= 1
                left += 1

            max_len = max(max_len, right - left + 1)

        return max_len
Time: O(n)
Space: O(1) — fixed 26-size array

76. Minimum Window Substring

Variable Window + Frequency

Given s and t, return the smallest substring of s containing every character of t (with counts), or "".

Python
from collections import Counter

class Solution:
    def minWindow(self, s, t):
        if not t or not s:
            return ""

        need = Counter(t)
        missing = len(t)
        left = 0
        best_left, best_right = 0, 0

        for right, char in enumerate(s, 1):
            if need[char] > 0:
                missing -= 1
            need[char] -= 1

            if missing == 0:
                while left < right and need[s[left]] < 0:
                    need[s[left]] += 1
                    left += 1

                if best_right == 0 or right - left < best_right - best_left:
                    best_left, best_right = left, right

                need[s[left]] += 1
                missing += 1
                left += 1

        return s[best_left:best_right]
Time: O(n + m)
Space: O(k) — distinct characters in t
