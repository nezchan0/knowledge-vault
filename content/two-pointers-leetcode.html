977. Squares of a Sorted Array

Two Pointers

Given nums sorted non-decreasing, return the squares of each number, also sorted non-decreasing.

Python
class Solution:
    def sortedSquares(self, nums):
        n = len(nums)
        result = [0] * n

        left, right = 0, n - 1
        pos = n - 1

        while left <= right:
            left_sq = nums[left] ** 2
            right_sq = nums[right] ** 2

            if left_sq > right_sq:
                result[pos] = left_sq
                left += 1
            else:
                result[pos] = right_sq
                right -= 1

            pos -= 1

        return result
Time: O(n)
Space: O(n) — output

392. Is Subsequence

Two Pointers

Return true if s is a subsequence of t.

Python
def is_subsequence(s, t):
    i = j = 0

    while i < len(s) and j < len(t):
        if s[i] == t[j]:
            i += 1
        j += 1

    return i == len(s)
Time: O(n)
Space: O(1)

283. Move Zeroes

Two Pointers

Move all 0s in nums to the end while keeping the relative order of non-zero elements, in-place.

Python
def move_zeroes(nums):
    insert_pos = 0

    for i in range(len(nums)):
        if nums[i] != 0:
            nums[insert_pos], nums[i] = nums[i], nums[insert_pos]
            insert_pos += 1
Time: O(n)

75. Sort Colors

Dutch National Flag

Given nums containing only 0, 1, 2, sort it in-place so same colors are adjacent, in order 0 → 1 → 2.

Python
class Solution:
    def sortColors(self, nums):
        low = mid = 0
        high = len(nums) - 1

        while mid <= high:
            if nums[mid] == 0:
                nums[low], nums[mid] = nums[mid], nums[low]
                low += 1
                mid += 1

            elif nums[mid] == 1:
                mid += 1

            else:
                nums[mid], nums[high] = nums[high], nums[mid]
                high -= 1
Time: O(n)
Space: O(1)

167. Two Sum II - Input Array Is Sorted

Two Pointers

Given a 1-indexed sorted array numbers, return the 1-based indices of two numbers that add up to target.

Python
class Solution:
    def twoSum(self, numbers, target):
        left, right = 0, len(numbers) - 1

        while left < right:
            total = numbers[left] + numbers[right]

            if total == target:
                return [left + 1, right + 1]

            elif total > target:
                right -= 1

            else:
                left += 1
Time: O(n)
Space: O(1)

15. 3Sum

Sort + Two Pointers

Given nums, return all unique triplets that sum to 0.

Python
class Solution:
    def threeSum(self, nums):
        ans = []
        nums.sort()

        for i in range(len(nums)):
            if i > 0 and nums[i] == nums[i - 1]:
                continue

            left, right = i + 1, len(nums) - 1

            while left < right:
                total = nums[i] + nums[left] + nums[right]

                if total == 0:
                    ans.append([nums[i], nums[left], nums[right]])

                    while left < right and nums[left] == nums[left + 1]:
                        left += 1

                    while left < right and nums[right] == nums[right - 1]:
                        right -= 1

                    left += 1
                    right -= 1

                elif total < 0:
                    left += 1
                else:
                    right -= 1

        return ans
Time: O(n²)
Space: O(1) — excluding output

16. 3Sum Closest

Sort + Two Pointers

Given nums and target, find three integers whose sum is closest to target and return that sum.

Python
class Solution:
    def threeSumClosest(self, nums, target):
        nums.sort()
        closest = float('inf')

        for i in range(len(nums)):
            if i > 0 and nums[i] == nums[i - 1]:
                continue

            left, right = i + 1, len(nums) - 1

            while left < right:
                curr_sum = nums[i] + nums[left] + nums[right]

                if abs(target - curr_sum) < abs(target - closest):
                    closest = curr_sum

                if curr_sum == target:
                    return curr_sum
                elif curr_sum < target:
                    left += 1
                else:
                    right -= 1

        return closest
Time: O(n²)
Space: O(1) — excluding output

18. 4Sum

Sort + Two Pointers

Given nums and target, return all unique quadruplets that sum to target.

Python
class Solution:
    def fourSum(self, nums, target):
        nums.sort()
        ans = []
        n = len(nums)

        for i in range(n):
            if i > 0 and nums[i] == nums[i - 1]:
                continue

            for j in range(i + 1, n):
                if j > i + 1 and nums[j] == nums[j - 1]:
                    continue

                left, right = j + 1, n - 1

                while left < right:
                    total = (
                        nums[i] + nums[j] +
                        nums[left] + nums[right]
                    )

                    if total == target:
                        ans.append([
                            nums[i], nums[j],
                            nums[left], nums[right]
                        ])

                        while left < right and nums[left] == nums[left + 1]:
                            left += 1

                        while left < right and nums[right] == nums[right - 1]:
                            right -= 1

                        left += 1
                        right -= 1

                    elif total < target:
                        left += 1
                    else:
                        right -= 1

        return ans
Time: O(n³)
Space: O(1) — excluding output

11. Container With Most Water

Two Pointers

Given height, find two lines that with the x-axis form the container holding the most water.

Python
class Solution:
    def maxArea(self, height):
        left, right = 0, len(height) - 1
        maxsum = 0

        while left <= right:
            h = min(height[left], height[right])
            currsum = h * (right - left)

            maxsum = max(maxsum, currsum)

            if height[left] < height[right]:
                left += 1
            else:
                right -= 1

        return maxsum
Time: O(n)
Space: O(1)

42. Trapping Rain Water

Two Pointers

Given an elevation map height of unit width, compute how much water it can trap after raining.

Python
class Solution:
    def trap(self, height):
        left, right = 0, len(height) - 1
        leftwall, rightwall = 0, 0
        total = 0

        while left < right:
            if height[left] < height[right]:
                leftwall = max(leftwall, height[left])
                total += leftwall - height[left]
                left += 1

            else:
                rightwall = max(rightwall, height[right])
                total += rightwall - height[right]
                right -= 1

        return total
Time: O(n)
Space: O(1)
