704. Binary Search

Classic BS

Given a sorted array nums and a target, return its index or -1 if not found.

Python
class Solution:
    def search(self, nums, target):
        low, high = 0, len(nums) - 1

        while low <= high:
            mid = (low + high) // 2

            if nums[mid] == target:
                return mid
            elif nums[mid] < target:
                low = mid + 1
            else:
                high = mid - 1

        return -1
Time: O(log n)
Space: O(1)

367. Valid Perfect Square

Classic BS

Given a positive integer num, determine whether it is a perfect square without using a built-in sqrt function.

Python
class Solution:
    def isPerfectSquare(self, num):
        low, high = 1, num

        while low <= high:
            mid = (low + high) // 2

            square = mid * mid

            if square == num:
                return True
            elif square < num:
                low = mid + 1
            else:
                high = mid - 1

        return False

74. Search a 2D Matrix

Flatten + BS

Given a matrix where each row is sorted and rows are sequentially sorted, return true if target exists, in O(log(m×n)).

Python
class Solution:
    def searchMatrix(self, matrix, target):
        m = len(matrix)
        n = len(matrix[0])

        low, high = 0, m * n - 1

        while low <= high:
            mid = (low + high) // 2

            row = mid // n
            col = mid % n

            if matrix[row][col] == target:
                return True
            elif matrix[row][col] < target:
                low = mid + 1
            else:
                high = mid - 1

        return False
Time: O(log(m × n))

240. Search a 2D Matrix II

Top-Right Search

Given a matrix sorted ascending along rows and columns, find whether target exists.

Python
class Solution:
    def searchMatrix(self, matrix, target):
        m = len(matrix)
        n = len(matrix[0])

        row = 0
        col = n - 1

        while row < m and col >= 0:
            if matrix[row][col] == target:
                return True

            elif matrix[row][col] > target:
                col -= 1

            else:
                row += 1

        return False
Time: O(m + n)

4. Median of Two Sorted Arrays

Partition BS

Given two sorted arrays nums1 and nums2, find their median in O(log(m + n)).

Python
class Solution:
    def findMedianSortedArrays(self, nums1, nums2):
        if len(nums1) > len(nums2):
            nums1, nums2 = nums2, nums1

        m, n = len(nums1), len(nums2)

        low, high = 0, m

        while low <= high:
            cut1 = (low + high) // 2
            cut2 = (m + n + 1) // 2 - cut1

            left1 = float('-inf') if cut1 == 0 else nums1[cut1 - 1]
            right1 = float('inf') if cut1 == m else nums1[cut1]

            left2 = float('-inf') if cut2 == 0 else nums2[cut2 - 1]
            right2 = float('inf') if cut2 == n else nums2[cut2]

            if left1 <= right2 and left2 <= right1:

                if (m + n) % 2 == 0:
                    return (
                        max(left1, left2) +
                        min(right1, right2)
                    ) / 2

                return max(left1, left2)

            elif left1 > right2:
                high = cut1 - 1

            else:
                low = cut1 + 1
Time: O(log(min(m, n)))
Space: O(1)

875. Koko Eating Bananas

BS on Answer

Given piles and h hours, find the minimum eating speed k to finish all piles within h hours.

Python
class Solution:
    def minEatingSpeed(self, piles, h):

        def is_possible(speed):
            hours = 0

            for pile in piles:
                hours += (pile + speed - 1) // speed

                if hours > h:
                    return False

            return True

        low, high = 1, max(piles)

        while low <= high:
            mid = (low + high) // 2

            if is_possible(mid):
                high = mid - 1
            else:
                low = mid + 1

        return low

410. Split Array Largest Sum

BS on Answer

Split nums into k contiguous subarrays minimizing the largest subarray sum.

Python
class Solution:
    def splitArray(self, nums, k):

        def is_possible(max_sum):
            parts = 1
            curr = 0

            for num in nums:
                if curr + num > max_sum:
                    parts += 1
                    curr = num
                else:
                    curr += num

            return parts <= k

        low = max(nums)
        high = sum(nums)

        while low <= high:
            mid = (low + high) // 2

            if is_possible(mid):
                high = mid - 1
            else:
                low = mid + 1

        return low
Time: O(n log(sum(nums)))
Space: O(1)

Given a sorted array and a target, return its first and last index, or [-1,-1] if absent.

class Solution:
    def searchRange(self, nums, target):

        def first_position():
            low, high = 0, len(nums) - 1
            result = -1

            while low <= high:
                mid = (low + high) // 2

                if nums[mid] == target:
                    result = mid
                    high = mid - 1
                elif nums[mid] < target:
                    low = mid + 1
                else:
                    high = mid - 1

            low, high = 0, len(nums) - 1

                    result = mid
                    low = mid + 1
                elif nums[mid] < target:
                    low = mid + 1
                else:
                    high = mid - 1

            return result

        return [first_position(), last_position()]

33. Search in Rotated Sorted Array

Rotated BS

A sorted array of distinct values has been rotated. Find the index of target, or -1.

Python
class Solution:
    def search(self, nums, target):
        low, high = 0, len(nums) - 1

        while low <= high:
            mid = (low + high) // 2

            if nums[mid] == target:
                return mid

            # Left half is sorted
            if nums[low] <= nums[mid]:

                if nums[low] <= target < nums[mid]:
                    high = mid - 1
                else:
                    low = mid + 1

            # Right half is sorted
            else:

                if nums[mid] < target <= nums[high]:
                    low = mid + 1
                else:
                    high = mid - 1

        return -1
Time: O(log n)
Space: O(1)

162. Find Peak Element

Peak BS

Find an index of a peak element — one greater than both neighbors.

Python
class Solution:
    def findPeakElement(self, nums):
        left, right = 0, len(nums) - 1

        while left < right:
            mid = (left + right) // 2

            if nums[mid] < nums[mid + 1]:
                left = mid + 1
            else:
                right = mid

        return left
