231. Power of Two

n & (n-1)

Given an integer n, return true if it is a power of two.

Python
class Solution:
    def isPowerOfTwo(self, n):
        return n > 0 and (n & (n - 1)) == 0

476. Number Complement

Bit Mask

Given a positive integer num, return its complement (flip all bits within its bit length).

Python
class Solution:
    def findComplement(self, num):
        mask = 1

        while mask < num:
            mask = (mask << 1) | 1

        return num ^ mask

190. Reverse Bits

Bit Reversal

Reverse the bits of a given 32-bit unsigned integer n.

Python
class Solution:
    def reverseBits(self, n):
        result = 0

        for _ in range(32):
            result = (result << 1) | (n & 1)
            n >>= 1

        return result
Time: O(1) — fixed 32 bits

504. Base 7

Base Conversion

Given an integer num, return its base 7 string representation.

Python
class Solution:
    def convertToBase7(self, num):
        if num == 0:
            return "0"

        negative = num < 0
        num = abs(num)
        digits = []

        while num:
            digits.append(str(num % 7))
            num //= 7

        if negative:
            digits.append('-')

        return ''.join(reversed(digits))
Time: O(log₇ n)
Space: O(log₇ n) — output

191. Number of 1 Bits

Brian Kernighan's

Given an integer n, return the number of set bits (Hamming weight) in its binary representation.

Python
class Solution:
    def hammingWeight(self, n):
        count = 0

        while n:
            n &= (n - 1)
            count += 1

        return count
Time: O(k) — k = number of set bits

268. Missing Number

XOR

Given nums containing n distinct numbers from 0 to n, return the one missing number.

Python
class Solution:
    def missingNumber(self, nums):
        n = len(nums)
        result = n

        for i, num in enumerate(nums):
            result ^= i ^ num

        return result

137. Single Number II

Bitmask State Machine

Given nums where every element appears three times except one, return that single element.

Python
class Solution:
    def singleNumber(self, nums):
        ones, twos = 0, 0

        for num in nums:
            ones = (ones ^ num) & ~twos
            twos = (twos ^ num) & ~ones

        return ones
Time: O(n)

260. Single Number III

XOR + Lowest Set Bit

Given nums where exactly two elements appear once and the rest appear twice, return the two single elements.

Python
class Solution:
    def singleNumber(self, nums):
        xor_all = 0

        for num in nums:
            xor_all ^= num

        diff = xor_all & (-xor_all)

        a = 0

        for num in nums:
            if num & diff:
                a ^= num

        b = xor_all ^ a

        return [a, b]
Time: O(n)

318. Maximum Product of Word Lengths

Bitmask

Given a string array words, return the maximum value of len(words[i]) * len(words[j]) for two words sharing no common letters.

Python
class Solution:
    def maxProduct(self, words):
        masks = []

        for word in words:
            mask = 0

            for c in word:
                mask |= 1 << (ord(c) - ord('a'))

            masks.append(mask)

        max_prod = 0
        n = len(words)

        for i in range(n):
            for j in range(i + 1, n):
                if masks[i] & masks[j] == 0:
                    max_prod = max(max_prod, len(words[i]) * len(words[j]))

        return max_prod
Time: O(n² + total chars)
Space: O(n)

371. Sum of Two Integers

Bit Carry Simulation

Given two integers a and b, return their sum without using + or -.

Python
class Solution:
    def getSum(self, a, b):
        mask = 0xFFFFFFFF

        while b != 0:
            carry = (a & b) << 1
            a = (a ^ b) & mask
            b = carry & mask

        return a if a <= 0x7FFFFFFF else ~(a ^ mask)
Time: O(1) — fixed 32-bit width
Space: O(1)

201. Bitwise AND of Numbers Range

Common Prefix Shift

Given left and right, return the bitwise AND of all numbers in [left, right].

Python
class Solution:
    def rangeBitwiseAnd(self, left, right):
        shift = 0

        while left < right:
            left >>= 1
            right >>= 1
            shift += 1

        return left << shift
Time: O(log n)
Space: O(1)
