832. Flipping an Image

Reverse + Invert

Given a binary image, flip each row horizontally, then invert it (0 ↔ 1); return the result.

Python
class Solution:
    def flipAndInvertImage(self, image):
        for row in image:
            row.reverse()

            for i in range(len(row)):
                row[i] ^= 1

        return image
Time: O(m × n)
Space: O(1) — in-place

867. Transpose Matrix

Transpose

Given a 2D matrix, return its transpose (swap rows and columns).

Python
class Solution:
    def transpose(self, matrix):
        m, n = len(matrix), len(matrix[0])
        result = [[0] * m for _ in range(n)]

        for i in range(m):
            for j in range(n):
                result[j][i] = matrix[i][j]

        return result
Time: O(m × n)
Space: O(m × n) — output

48. Rotate Image

Transpose + Reverse

Rotate an n × n matrix 90 degrees clockwise, in-place.

Python
class Solution:
    def rotate(self, matrix):
        n = len(matrix)

        for i in range(n):
            for j in range(i + 1, n):
                matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j]

        for row in matrix:
            row.reverse()
Time: O(n²)
Space: O(1)

1572. Matrix Diagonal Sum

Diagonal Traversal

Given a square matrix mat, return the sum of both diagonals, without double-counting the center on odd-sized matrices.

Python
class Solution:
    def diagonalSum(self, mat):
        n = len(mat)
        total = 0

        for i in range(n):
            total += mat[i][i]

            if i != n - 1 - i:
                total += mat[i][n - 1 - i]

        return total
Time: O(n)

Reshape mat into an r × c matrix if the element count matches; otherwise return the original.

Python
class Solution:
    def matrixReshape(self, mat, r, c):
        m, n = len(mat), len(mat[0])

        if m * n != r * c:
            return mat

        flat = [mat[i][j] for i in range(m) for j in range(n)]

        return [flat[i * c:(i + 1) * c] for i in range(r)]

118. Pascal's Triangle

Row Build

Given numRows, generate the first numRows of Pascal's Triangle.

Python
class Solution:
    def generate(self, numRows):
        triangle = []

        for i in range(numRows):
            row = [1] * (i + 1)

            for j in range(1, i):
                row[j] = triangle[i - 1][j - 1] + triangle[i - 1][j]

            triangle.append(row)

        return triangle
Time: O(numRows²)

766. Toeplitz Matrix

Diagonal Check

Return true if every diagonal from top-left to bottom-right has the same value throughout.

Python
class Solution:
    def isToeplitzMatrix(self, matrix):
        m, n = len(matrix), len(matrix[0])

        for i in range(1, m):
            for j in range(1, n):
                if matrix[i][j] != matrix[i - 1][j - 1]:
                    return False

        return True

73. Set Matrix Zeroes

In-Place Markers

If an element of matrix is 0, set its entire row and column to 0, in-place.

Python
class Solution:
    def setZeroes(self, matrix):
        m, n = len(matrix), len(matrix[0])

        first_row_has_zero = any(matrix[0][j] == 0 for j in range(n))
        first_col_has_zero = any(matrix[i][0] == 0 for i in range(m))

        for i in range(1, m):
            for j in range(1, n):
                if matrix[i][j] == 0:
                    matrix[i][0] = 0
                    matrix[0][j] = 0

        for i in range(1, m):
            for j in range(1, n):
                if matrix[i][0] == 0 or matrix[0][j] == 0:
                    matrix[i][j] = 0

        if first_row_has_zero:
            for j in range(n):
                matrix[0][j] = 0

        if first_col_has_zero:
            for i in range(m):
                matrix[i][0] = 0
Time: O(m × n)
Space: O(1)

498. Diagonal Traverse

Diagonal Zigzag

Given an m × n matrix mat, return all elements in diagonal zigzag order.

Python
class Solution:
    def findDiagonalOrder(self, mat):
        m, n = len(mat), len(mat[0])
        result = []
        row, col = 0, 0
        going_up = True

        for _ in range(m * n):
            result.append(mat[row][col])

            if going_up:
                if col == n - 1:
                    row += 1
                    going_up = False
                elif row == 0:
                    col += 1
                    going_up = False
                else:
                    row -= 1
                    col += 1
            else:
                if row == m - 1:
                    col += 1
                    going_up = True
                elif col == 0:
                    row += 1
                    going_up = True
                else:
                    row += 1
                    col -= 1

        return result
Time: O(m × n)
Space: O(1) — excluding output

54. Spiral Matrix

Boundary Traversal

Given an m × n matrix, return all elements in spiral order.

Python
class Solution:
    def spiralOrder(self, matrix):
        left, right, top, down = 0, len(matrix[0]) - 1, 0, len(matrix) - 1
        ans = []

        while left <= right and top <= down:

            for a in range(left, right + 1):
                ans.append(matrix[top][a])
            top += 1

            for b in range(top, down + 1):
                ans.append(matrix[b][right])
            right -= 1

            if top <= down:
                for c in range(right, left - 1, -1):
                    ans.append(matrix[down][c])
                down -= 1

            if left <= right:
                for d in range(down, top - 1, -1):
                    ans.append(matrix[d][left])
                left += 1

        return ans
Time: O(m × n)
Space: O(m × n) — output

59. Spiral Matrix II

Boundary Fill

Given an integer n, generate an n × n matrix filled with 1 to n² in spiral order.

Python
class Solution:
    def generateMatrix(self, n):
        matrix = [[0] * n for _ in range(n)]
        left, right, top, down = 0, n - 1, 0, n - 1
        num = 1

        while left <= right and top <= down:
            for col in range(left, right + 1):
                matrix[top][col] = num
                num += 1
            top += 1

            for row in range(top, down + 1):
                matrix[row][right] = num
                num += 1
            right -= 1

            for col in range(right, left - 1, -1):
                matrix[down][col] = num
                num += 1
            down -= 1

            for row in range(down, top - 1, -1):
                matrix[row][left] = num
                num += 1
            left += 1

        return matrix
Time: O(n²)
Space: O(n²) — output

240. Search a 2D Matrix II

Top-Right Search

Given a matrix sorted ascending along rows and columns, return true if target exists.

Python
class Solution:
    def searchMatrix(self, matrix, target):
        m, n = len(matrix), len(matrix[0])

        row, col = 0, n - 1

        while row < m and col >= 0:
            if matrix[row][col] == target:
                return True

            elif matrix[row][col] > target:
                col -= 1

            else:
                row += 1

        return False
Time: O(m + n)
Space: O(1)
