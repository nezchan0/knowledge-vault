Given jewels and stones, return how many characters in stones are also in jewels (case-sensitive).

        jewels_set = set(jewels)

class Solution:
    def canConstruct(self, ransomNote: str, magazine: str) -> bool:
        freq = [0] * 26

        for c in magazine:
            freq[ord(c) - ord('a')] += 1

        for c in ransomNote:
            freq[ord(c) - ord('a')] -= 1

        return True

class Solution:
    def isAnagram(self, s: str, t: str) -> bool:
        if len(s) != len(t):

            freq[ord(c) - ord('a')] += 1

        return all(x == 0 for x in freq)

class Solution:

2287. Rearrange Characters to Make Target String

Frequency Array / Counter

Return the maximum number of copies of target that can be formed using the characters of s, each used at most once.

Frequency Array
class Solution:
    def rearrangeCharacters(self, s: str, target: str) -> int:
        sfreq = [0] * 26
        tfreq = [0] * 26

        for c in s:
            sfreq[ord(c) - ord('a')] += 1

        for c in target:
            tfreq[ord(c) - ord('a')] += 1

        ans = float('inf')

        for i in range(26):
            if tfreq[i] > 0:
                ans = min(ans, sfreq[i] // tfreq[i])

        return ans
Time: O(n + m)
Space: O(1)
Counter
from collections import Counter

class Solution:
    def rearrangeCharacters(self, s: str, target: str) -> int:
        s_count = Counter(s)
        t_count = Counter(target)

        return min(
            s_count[c] // t_count[c]
            for c in t_count
        )

1. Two Sum

HashMap

Return indices of the two numbers in nums that add up to target.

Python
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        numset = {}

        for i, ele in enumerate(nums):
            comp = target - ele

            if comp in numset:
                return [numset[comp], i]

            numset[ele] = i

class Solution:
    def majorityElement(self, nums: List[int]) -> int:
        count = 0
        candidate = None

        for ele in nums:
            if count == 0:
                candidate = ele

            count += 1 if ele == candidate else -1

        return candidate

36. Valid Sudoku

Multiple Sets

Determine if a 9×9 Sudoku board is valid — no repeated digits per row, column, or 3×3 box.

Python
class Solution:
    def isValidSudoku(self, board: List[List[str]]) -> bool:
        rowset = [set() for _ in range(9)]
        colset = [set() for _ in range(9)]
        boxset = [set() for _ in range(9)]

        for r in range(9):
            for c in range(9):
                val = board[r][c]

                if val == ".":
                    continue

                boxindex = (r // 3) * 3 + (c // 3)

                if (val in rowset[r] or
                    val in colset[c] or
                    val in boxset[boxindex]):
                    return False

                rowset[r].add(val)
                colset[c].add(val)
                boxset[boxindex].add(val)

        return True
Time: O(81) → O(1)
Space: O(81) → O(1)

49. Group Anagrams

HashMap + Key

Given an array of strings strs, group the anagrams together.

Python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        dic = {}

        for ele in strs:
            key = "".join(sorted(ele))

            if key not in dic:
                dic[key] = []

            dic[key].append(ele)

        return list(dic.values())
Time: O(n × k log k)

128. Longest Consecutive Sequence

Set

Given an unsorted array nums, return the length of the longest consecutive elements sequence in O(n).

Python
class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        nset = set(nums)
        longest = 0

        for ele in nset:
            if ele - 1 not in nset:
                length = 1

                while ele + length in nset:
                    length += 1

                longest = max(longest, length)

        return longest
Time: O(n) average
Space: O(n)

560. Subarray Sum Equals K

Prefix Sum + HashMap

Return the total number of contiguous subarrays of nums whose sum equals k.

Python
from collections import defaultdict

class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        prefix = 0
        count = 0

        freq = defaultdict(int)
        freq[0] = 1

        for num in nums:
            prefix += num

            if prefix - k in freq:
                count += freq[prefix - k]

            freq[prefix] += 1

        return count
Time: O(n)
Space: O(n)
