83. Remove Duplicates from Sorted List

Pointer

Given a sorted linked list, delete duplicates so each element appears only once.

Python
def deleteDuplicates(self, head):
    if head is None:
        return None

    curr = head

    while curr.next:
        if curr.val == curr.next.val:
            curr.next = curr.next.next
        else:
            curr = curr.next

    return head
Time: O(n)
Space: O(1)

2. Add Two Numbers

Carry / Simulation

Two non-empty lists represent non-negative integers with digits in reverse order. Add them and return the sum as a linked list.

Python
def addTwoNumbers(self, l1, l2):
    dummy = ListNode(0)
    curr = dummy
    carry = 0

    while l1 or l2 or carry:
        x = l1.val if l1 else 0
        y = l2.val if l2 else 0

        total = x + y + carry

        carry = total // 10
        digit = total % 10

        curr.next = ListNode(digit)
        curr = curr.next

        if l1:
            l1 = l1.next
        if l2:
            l2 = l2.next

    return dummy.next
Time: O(max(n, m))
Space: O(max(n, m)) — output

21. Merge Two Sorted Lists

Two Pointers

Merge two sorted linked lists list1 and list2 into one sorted list.

Python
def mergeTwoLists(self, list1, list2):
    dummy = ListNode(0)
    curr = dummy

    while list1 and list2:
        if list1.val < list2.val:
            curr.next = list1
            list1 = list1.next
        else:
            curr.next = list2
            list2 = list2.next

        curr = curr.next

    if list1:
        curr.next = list1
    else:
        curr.next = list2

    return dummy.next
Time: O(n + m)
Space: O(1)

148. Sort List

Merge Sort

Given the head of a linked list, return it sorted in ascending order.

Python
def sortList(self, head):
    if not head or not head.next:
        return head

    # Find middle
    slow = head
    fast = head.next

    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next

    mid = slow.next
    slow.next = None

    # Sort both halves
    left = self.sortList(head)
    right = self.sortList(mid)

    # Merge
    dummy = ListNode(0)
    curr = dummy

    while left and right:
        if left.val < right.val:
            curr.next = left
            left = left.next
        else:
            curr.next = right
            right = right.next

        curr = curr.next

    curr.next = left if left else right

    return dummy.next
Time: O(n log n)
Space: O(log n) — recursion stack

2807. Insert Greatest Common Divisors in Linked List

GCD + Pointer

Between every pair of adjacent nodes, insert a new node containing the GCD of those two values.

Python
def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)


def insertGreatestCommonDivisors(self, head):
    if head is None or head.next is None:
        return head

    curr = head

    while curr.next:
        value = gcd(curr.val, curr.next.val)

        new_node = ListNode(value)
        new_node.next = curr.next
        curr.next = new_node

        curr = new_node.next

    return head
Time: O(n log V)
Space: O(log V) — recursive GCD

19. Remove Nth Node From End of List

Two Pointers

Given the head of a linked list, remove the nth node from the end and return the head.

Python
def removeNthFromEnd(self, head, n):
    dummy = ListNode(0)
    dummy.next = head

    slow = dummy
    fast = dummy

    for _ in range(n):
        fast = fast.next

    while fast.next:
        slow = slow.next
        fast = fast.next

    slow.next = slow.next.next

    return dummy.next

138. Copy List with Random Pointer

Interleaving

Each node has val, next, and random pointers. Return a deep copy of the list.

Python
def copyRandomList(self, head):
    if not head:
        return None

    # 1. Create copied nodes
    curr = head

    while curr:
        copy = Node(curr.val)

        copy.next = curr.next
        curr.next = copy

        curr = copy.next

    # 2. Assign random pointers
    curr = head

    while curr:
        if curr.random:
            curr.next.random = curr.random.next

        curr = curr.next.next

    # 3. Separate lists
    curr = head
    copy_head = head.next

    while curr:
        copy = curr.next

        curr.next = copy.next

        if copy.next:
            copy.next = copy.next.next

        curr = curr.next

    return copy_head
Time: O(n)
Space: O(1) — excluding copied nodes

141. Linked List Cycle

def hasCycle(self, head):
    slow = head
    fast = head

    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next

        if slow == fast:
            return True

    return False

142. Linked List Cycle II

Floyd's (Slow/Fast)

If the linked list has a cycle, return the node where the cycle begins; otherwise return None.

Python
def detectCycle(self, head):
    slow = head
    fast = head

    # Detect cycle
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next

        if slow == fast:
            break
    else:
        return None

    # Find cycle start
    slow = head

    while slow != fast:
        slow = slow.next
        fast = fast.next

    return slow
Time: O(n)
Space: O(1)

160. Intersection of Two Linked Lists

Length Alignment

Given two singly linked lists, return the node where they intersect, or None if they don't.

Python
def getIntersectionNode(self, headA, headB):
    def length(head):
        count = 0

        while head:
            count += 1
            head = head.next

        return count

    lenA = length(headA)
    lenB = length(headB)

    while lenA > lenB:
        headA = headA.next
        lenA -= 1

    while lenB > lenA:
        headB = headB.next
        lenB -= 1

    while headA != headB:
        headA = headA.next
        headB = headB.next

    return headA
Time: O(n + m)
Space: O(1)

206. Reverse Linked List

prev / curr / next

Given the head of a singly linked list, reverse the list and return the new head.

Python
def reverseList(self, head):
    prev = None
    curr = head

    while curr:
        nextnode = curr.next
        curr.next = prev

        prev = curr
        curr = nextnode

    return prev

234. Palindrome Linked List

Reverse Half

Determine whether a linked list reads the same forwards and backwards.

Python
def isPalindrome(self, head):
    if not head or not head.next:
        return True

    # Find middle
    slow = head
    fast = head

    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next

    # Reverse second half
    prev = None

    while slow:
        nextnode = slow.next
        slow.next = prev
        prev = slow
        slow = nextnode

    # Compare
    left = head
    right = prev

    while right:
        if left.val != right.val:
            return False

        left = left.next
        right = right.next

    return True
Time: O(n)
Space: O(1)

92. Reverse Linked List II

Partial Reverse

Reverse the linked list nodes from position left to position right, in one pass.

Python
def reverseBetween(self, head, left, right):
    dummy = ListNode(0)
    dummy.next = head

    prev = dummy

    # Move prev to node before left
    for _ in range(left - 1):
        prev = prev.next

    curr = prev.next

    # Reverse section
    for _ in range(right - left):
        temp = curr.next

        curr.next = temp.next
        temp.next = prev.next
        prev.next = temp

    return dummy.next
Time: O(n)
Space: O(1)

25. Reverse Nodes in k-Group

Group Reversal

Reverse the nodes of a linked list k at a time; leave a final group of fewer than k nodes unchanged.

Python
def reverseKGroup(self, head, k):
    dummy = ListNode(0)
    dummy.next = head

    group_prev = dummy

    while True:
        # Find kth node
        kth = group_prev

        for _ in range(k):
            kth = kth.next

            if not kth:
                return dummy.next

        group_next = kth.next

        # Reverse group
        prev = group_next
        curr = group_prev.next

        while curr != group_next:
            nextnode = curr.next
            curr.next = prev
            prev = curr
            curr = nextnode

        # Connect previous group
        temp = group_prev.next
        group_prev.next = kth
        group_prev = temp
Time: O(n)
Space: O(1)
