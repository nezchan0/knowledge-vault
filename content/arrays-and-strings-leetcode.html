2239. Find Closest Number to Zero

Traversal

Return the number in nums closest to 0; if tied, return the larger one.

Python
class Solution:
    def findClosestNumber(self, nums: List[int]) -> int:
        closest = float('inf')

        for ele in nums:
            if abs(ele) == abs(closest):
                closest = max(closest, ele)
            elif abs(ele) < abs(closest):
                closest = ele

        return closest
Time: O(n)
Space: O(1)

1768. Merge Strings Alternately

Two Pointers

Merge word1 and word2 by alternating characters, starting with word1; append leftover characters at the end.

Python
class Solution:
    def mergeAlternately(self, word1: str, word2: str) -> str:
        n1 = len(word1)
        n2 = len(word2)

        ans = []
        i = 0

        while i < min(n2, n1):
            ans.append(word1[i])
            ans.append(word2[i])
            i += 1

        ans.append(word1[i:])
        ans.append(word2[i:])

        return "".join(ans)
Time: O(n + m)

13. Roman to Integer

Hash Map

Convert a Roman numeral string s into an integer.

Python
class Solution:
    def romanToInt(self, s: str) -> int:
        dic = {
            "I": 1,
            "V": 5,
            "X": 10,
            "L": 50,
            "C": 100,
            "D": 500,
            "M": 1000
        }

        sum = dic[s[0]]

        for i in range(1, len(s)):
            sum += dic[s[i]]

            if s[i-1] == "I" and (s[i] == "V" or s[i] == "X"):
                sum -= 2 * dic["I"]

            elif s[i-1] == "X" and (s[i] == "L" or s[i] == "C"):
                sum -= 2 * dic["X"]

            elif s[i-1] == "C" and (s[i] == "D" or s[i] == "M"):
                sum -= 2 * dic["C"]

        return sum
Time: O(n)
Space: O(1)

392. Is Subsequence

Two Pointers

Return true if s is a subsequence of t.

Python
class Solution:
    def isSubsequence(self, s: str, t: str) -> bool:
        i = 0
        j = 0

        while i < len(s) and j < len(t):
            if s[i] == t[j]:
                i += 1

            j += 1

        if i == len(s):
            return True

        return False
Time: O(n)

121. Best Time to Buy and Sell Stock

Running Min

Given daily prices, buy on one day and sell on a later day to maximize profit; return 0 if no profit is possible.

Python
class Solution:
    def maxProfit(self, prices: List[int]) -> int:
        minprice = float('inf')
        profit = 0
        n = len(prices)

        for i in range(n):
            if prices[i] < minprice:
                minprice = prices[i]

            profit = max(profit, prices[i] - minprice)

        return profit
Time: O(n)
Space: O(1)

14. Longest Common Prefix

String Compare

Find the longest common prefix string among an array of strings; return "" if none.

Python
class Solution:
    def longestCommonPrefix(self, strs: List[str]) -> str:

        if not strs:
            return ""

        for i in range(len(strs[0])):
            char = strs[0][i]

            for ele in strs[1:]:
                if i >= len(ele) or ele[i] != char:
                    return strs[0][:i]

        return strs[0]

228. Summary Ranges

Traversal

Given a sorted unique integer array nums, return the smallest list of ranges that exactly covers all numbers.

Python
class Solution:
    def summaryRanges(self, nums: List[int]) -> List[str]:
        if len(nums) == 0:
            return []

        rangearr = []
        prev = nums[0]
        nex = nums[0]

        for i in range(0, len(nums) - 1):
            nex = nums[i]

            if nums[i + 1] - nums[i] == 1:
                if prev == nex:
                    rangearr.append(f"{prev}")
                else:
                    rangearr.append(f"{prev}->{nex}")

                prev = nums[i + 1]

        nex = nums[len(nums) - 1]

        if prev == nex:
            rangearr.append(f"{prev}")
        else:
            rangearr.append(f"{prev}->{nex}")

        return rangearr
Time: O(n)
Space: O(1) — excluding output

238. Product of Array Except Self

Prefix / Suffix

Return answer where answer[i] is the product of all elements except nums[i], in O(n) time without division.

Python
class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        n = len(nums)
        answer = [1] * n

        # Product of all elements to the left
        prefix = 1

        for i in range(n):
            answer[i] = prefix
            prefix *= nums[i]

        # Product of all elements to the right
        suffix = 1

        for i in range(n - 1, -1, -1):
            answer[i] *= suffix
            suffix *= nums[i]

        return answer
Time: O(n)
Space: O(1) — excluding output

56. Merge Intervals

Sorting

Merge all overlapping intervals and return the non-overlapping intervals covering all input intervals.

Python
class Solution:
    def merge(self, intervals: List[List[int]]) -> List[List[int]]:
        intervals.sort(key=lambda x: x[0])

        ansarr = []
        index = 0
        currmerger = intervals[0]

        while index < len(intervals):
            if currmerger[1] >= intervals[index][0]:
                currmerger[1] = max(
                    currmerger[1],
                    intervals[index][1]
                )
                currmerger[0] = min(
                    currmerger[0],
                    intervals[index][0]
                )
            else:
                ansarr.append(currmerger)
                currmerger = intervals[index]

            index += 1

        ansarr.append(currmerger)

        return ansarr
Time: O(n log n)
Space: O(n)

54. Spiral Matrix

Boundary Traversal

Given an m × n matrix, return all elements in spiral order.

Python
class Solution:
    def spiralOrder(self, matrix: List[List[int]]) -> List[int]:
        left, right, top, down = 0, len(matrix[0]) - 1, 0, len(matrix) - 1

        a, b, c, d = 0, 0, 0, 0
        ansarr = []

        while left <= right and top <= down:

            for a in range(left, right + 1):
                ansarr.append(matrix[top][a])
            top += 1

            for b in range(top, down + 1):
                ansarr.append(matrix[b][right])
            right -= 1

            if top <= down:
                for c in range(right, left - 1, -1):
                    ansarr.append(matrix[down][c])
                down -= 1

            if left <= right:
                for d in range(down, top - 1, -1):
                    ansarr.append(matrix[d][left])
                left += 1

        return ansarr
Time: O(m × n)
Space: O(m × n) — output

6. Zigzag Conversion

Simulation

Write string s in a zigzag pattern across numRows rows, then read row by row.

Python
class Solution:
    def convert(self, s: str, numRows: int) -> str:
        if numRows == 1 or numRows >= len(s):
            return s

        rows = [""] * numRows
        curr_row = 0
        goingdown = False

        for char in s:
            rows[curr_row] += char

            if curr_row == 0 or curr_row == numRows - 1:
                goingdown = not goingdown

            curr_row += 1 if goingdown else -1

        return "".join(rows)
Time: O(n)
Space: O(n)
